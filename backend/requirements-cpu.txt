# ----------------------------------------------------------------------
# BAIF Bhasha - CPU build (default)
#
# Pulls the base deps from requirements.txt, then a plain CPU-only torch
# build from the normal PyPI index -- no NVIDIA driver dependency at all,
# so it runs on any Windows machine, GPU or not. This is what step 2 of
# PACKAGING.md installs by default.
#
# For a machine with a confirmed NVIDIA GPU + driver, use
# requirements-gpu.txt instead (see PACKAGING.md step 6b) -- don't install
# both into the same venv.
#
# The Docker image uses requirements-docker-cpu.txt, which pins the same
# torch release but pulls the Linux CPU-only wheel from PyTorch's index --
# on Linux (unlike Windows) plain PyPI torch hard-depends on the NVIDIA CUDA
# stack. Bump the versions in both files together.
# ----------------------------------------------------------------------

-r requirements.txt

torch==2.13.0
torchaudio==2.11.0
